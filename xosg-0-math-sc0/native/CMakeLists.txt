cmake_minimum_required(VERSION 3.16)
project(xosg_math_sc0_native LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
add_library(sc0native SHARED src/sc0_reduce.c src/sc0_native.cpp)
target_compile_definitions(sc0native PRIVATE SC0_NATIVE_BUILD)
target_include_directories(sc0native PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
add_executable(native_selftest tests/native_selftest.cpp)
target_link_libraries(native_selftest PRIVATE sc0native)
target_include_directories(native_selftest PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/include)
